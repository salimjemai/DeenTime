// Fixture generator for backend-node/src/domain/prayer-times.spec.ts.
//
// Runs the real DeenTime.Core IsnaCalculator over a matrix of calculation
// methods, Asr juristic methods, locations and dates and prints the results as
// JSON (System.Text.Json, camelCase), so the TypeScript port can be asserted
// against the .NET implementation value for value. The "expected" objects are
// the serialized PrayerTimesDto records, i.e. exactly what the .NET API emits.
//
// Build/run (outside the repo, referencing the Core/Contracts projects):
//   DOTNET_ROLL_FORWARD=Major dotnet run -c Release > prayer-times.json
using System.Text.Json;
using System.Text.Json.Serialization;
using DeenTime.Core.Entities;
using DeenTime.Core.Services;

var methods = new[]
{
    "ISNA", "MWL", "Muslim World League", "Karachi", "UmmAlQura", "Egyptian",
    "Gulf", "Kuwait", "Qatar", "Tehran", "Jafari", "nonsense",
};
var asrMethods = new[] { "Hanafi", "Other" };
var dates = new[]
{
    new DateOnly(2026, 1, 15),
    new DateOnly(2026, 3, 8),
    new DateOnly(2026, 6, 21),
    new DateOnly(2026, 11, 1),
    new DateOnly(2026, 12, 31),
};

var locations = new Dictionary<string, Location>
{
    // The Cedar Park fixture from PrayerTimeCalculatorTests.cs (IANA DST conversion).
    ["cedarPark"] = new(30.5052m, -97.8203m, "America/Chicago", true, null, null, 5, 1),
    // Same place, custom DST window: base offset + 1h while dstBegins <= date < dstEnds.
    ["cedarParkCustomDst"] = new(30.5052m, -97.8203m, "America/Chicago", true, new DateOnly(2026, 3, 8), new DateOnly(2026, 11, 1), 5, 1),
    // Same place, DST not observed: base (standard) offset all year.
    ["cedarParkNoDst"] = new(30.5052m, -97.8203m, "America/Chicago", false, null, null, 5, 1),
    // High latitude: the Fajr/Isha hour angle clamps kick in around the June solstice.
    ["anchorage"] = new(61.2181m, -149.9003m, "America/Anchorage", true, null, null, 2, 3),
    // No DST at all; solar noon is late in the UTC day so Isha spills into the next UTC day.
    ["honolulu"] = new(21.3069m, -157.8583m, "Pacific/Honolulu", false, null, null, 0, 0),
};

var calculator = new IsnaCalculator();
var cases = new List<Case>();
foreach (var (key, loc) in locations)
{
    foreach (var method in methods)
    {
        foreach (var asr in asrMethods)
        {
            foreach (var date in dates)
            {
                var criteria = new PrayerTimingCriteria
                {
                    Method = method,
                    JuristicMethodAsr = asr,
                    Latitude = loc.Latitude,
                    Longitude = loc.Longitude,
                    TimezoneId = loc.TimezoneId,
                    DstObserved = loc.DstObserved,
                    DstBegins = loc.DstBegins,
                    DstEnds = loc.DstEnds,
                    MinutesAfterZawal = loc.MinutesAfterZawal,
                    MinutesAfterMaghrib = loc.MinutesAfterMaghrib,
                };
                cases.Add(new Case(key, method, asr, calculator.Compute(criteria, date)));
            }
        }
    }
}

// TimeZoneInfo.BaseUtcOffset as seen by .NET for every zone used above.
var timeZones = locations.Values
    .Select(l => l.TimezoneId)
    .Distinct()
    .ToDictionary(id => id, id => new TimeZoneProbe((int)TimeZoneInfo.FindSystemTimeZoneById(id).BaseUtcOffset.TotalMinutes));

// DateTime.AddHours(double) precision probe: ticks added relative to midnight.
var midnight = new DateTime(2026, 8, 19, 0, 0, 0, DateTimeKind.Utc);
var probeHours = new[]
{
    0.0, 1e-7, -1e-7, 0.5, 5.5, -3.25, 12.123456789, 23.99999999999, 30.999999999,
    -0.000001, 11.833333333333334, 20.166666666666668, 26.749999999999996, 100.0 / 3.0,
};
var addHoursProbe = probeHours
    .Select(h => new AddHoursProbe(h, midnight.AddHours(h).Ticks - midnight.Ticks))
    .ToList();

var options = new JsonSerializerOptions
{
    PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
    WriteIndented = true,
    DefaultIgnoreCondition = JsonIgnoreCondition.Never,
};
var output = new Fixture(
    "Generated by prayer-times.generator.cs.txt against DeenTime.Core.Services.IsnaCalculator; do not edit by hand.",
    Environment.Version.ToString(),
    locations,
    timeZones,
    cases,
    addHoursProbe);
Console.Out.Write(JsonSerializer.Serialize(output, options));
Console.Out.WriteLine();

sealed record Location(
    decimal Latitude,
    decimal Longitude,
    string TimezoneId,
    bool DstObserved,
    DateOnly? DstBegins,
    DateOnly? DstEnds,
    int MinutesAfterZawal,
    int MinutesAfterMaghrib);

sealed record Case(string Location, string Method, string JuristicMethodAsr, DeenTime.Contracts.Timings.PrayerTimesDto Expected);

sealed record TimeZoneProbe(int BaseUtcOffsetMinutes);

sealed record AddHoursProbe(double Hours, long Ticks);

sealed record Fixture(
    string Comment,
    string DotnetVersion,
    Dictionary<string, Location> Locations,
    Dictionary<string, TimeZoneProbe> TimeZones,
    List<Case> Cases,
    List<AddHoursProbe> AddHoursProbe);
