static string Iso(DateTime dt) => $"{dt.Year:D4}-{dt.Month:D2}-{dt.Day:D2}";
static string Ymd(DateTime dt) => $"{dt.Year}-{dt.Month}-{dt.Day}";
string Hijri(DateTime dt) => $"{cal.GetYear(dt)}-{cal.GetMonth(dt)}-{cal.GetDayOfMonth(dt)}";

// (a) GetYear/GetMonth/GetDayOfMonth for every day 2019-12-01 .. 2036-12-31.
var monthStarts = new Dictionary<string, string>();
var days2025To2027 = new Dictionary<string, string>();
var perYearLines = new SortedDictionary<int, StringBuilder>();
var perYearCount = new SortedDictionary<int, int>();
var total = 0;
for (var dt = new DateTime(2019, 12, 1); dt <= new DateTime(2036, 12, 31); dt = dt.AddDays(1))
{
    var h = Hijri(dt);
    total++;
    if (dt.Day == 1) monthStarts[Iso(dt)] = h;
    if (dt.Year is >= 2025 and <= 2027) days2025To2027[Iso(dt)] = h;
    if (!perYearLines.TryGetValue(dt.Year, out var sb))
    {
        sb = new StringBuilder();
        perYearLines[dt.Year] = sb;
        perYearCount[dt.Year] = 0;
    }
    if (sb.Length > 0) sb.Append('\n');
    sb.Append(Iso(dt)).Append('=').Append(h);
    perYearCount[dt.Year]++;
}
var yearDigests = new Dictionary<string, object>();
foreach (var (year, sb) in perYearLines)
{
    yearDigests[year.ToString(CultureInfo.InvariantCulture)] = new
    {
        days = perYearCount[year],
        sha256 = Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(sb.ToString()))).ToLowerInvariant()
    };
}

// (d) HijriService.Generate replica for 2026-01 .. 2026-12.
List<Anchor> Generate(DateOnly from, DateOnly to)
{
    var months = new List<Anchor>();
    var cursor = new DateOnly(from.Year, from.Month, 1);
    var end = new DateOnly(to.Year, to.Month, 1);
    while (cursor <= end)
    {
        var gregorianFirst = cursor.ToDateTime(TimeOnly.MinValue);
        months.Add(new Anchor(cursor.Year, cursor.Month, cal.GetDayOfMonth(gregorianFirst), cal.GetMonth(gregorianFirst), cal.GetYear(gregorianFirst)));
        cursor = cursor.AddMonths(1);
    }
    return months;
}

// QuestPdfGenerator.GetHijriDate replica.
(int Day, int Month, int Year) GetHijriDate(DateOnly date, IReadOnlyDictionary<(int Year, int Month), Anchor> maps)
{
    DateTime current;
    if (maps.TryGetValue((date.Year, date.Month), out var map) && map.HijriMonthOnFirst > 0 && map.HijriYearOnFirst > 1)
    {
        var first = cal.ToDateTime(map.HijriYearOnFirst, map.HijriMonthOnFirst, map.HijriDayOnFirst, 0, 0, 0, 0);
        current = first.AddDays(date.Day - 1);
    }
    else
    {
        current = date.ToDateTime(TimeOnly.MinValue);
    }
    return (cal.GetDayOfMonth(current), cal.GetMonth(current), cal.GetYear(current));
}

// PublicController.BuildHijriDate replica.
object? BuildHijriDate(Anchor? map, DateOnly gregorianDate)
{
    string[] monthNames =
    [
        "Muharram", "Safar", "Rabi al-Awwal", "Rabi al-Thani",
        "Jumada al-Awwal", "Jumada al-Thani", "Rajab", "Sha'ban",
        "Ramadan", "Shawwal", "Dhu al-Qi'dah", "Dhu al-Hijjah"
    ];
    try
    {
        var hijriBase = map is not null && map.HijriMonthOnFirst > 0 && map.HijriYearOnFirst > 1
            ? cal.ToDateTime(map.HijriYearOnFirst, map.HijriMonthOnFirst, map.HijriDayOnFirst, 0, 0, 0, 0)
            : new DateTime(gregorianDate.Year, gregorianDate.Month, 1);
        var current = hijriBase.AddDays(gregorianDate.Day - 1);
        var day = cal.GetDayOfMonth(current);
        var month = cal.GetMonth(current);
        var year = cal.GetYear(current);
        var monthName = monthNames[month - 1];
        return new { day, month, year, monthName, formatted = $"{monthName} {day}, {year}" };
    }
    catch (ArgumentOutOfRangeException)
    {
        return null;
    }
}

var generated2026 = Generate(new DateOnly(2026, 1, 1), new DateOnly(2026, 12, 31));
var generated2026Maps = generated2026.ToDictionary(a => (a.Year, a.Month));
var empty = new Dictionary<(int Year, int Month), Anchor>();

// "Shifted" anchors: the Gregorian 1st is declared one Hijri day later than the tabular value
// (moon sighted a day early). Every anchor is a real Hijri date, so ToDateTime never throws.
var shifted2026 = new List<Anchor>();
for (var m = 1; m <= 12; m++)
{
    var next = new DateTime(2026, m, 2);
    shifted2026.Add(new Anchor(2026, m, cal.GetDayOfMonth(next), cal.GetMonth(next), cal.GetYear(next)));
}
var shifted2026Maps = shifted2026.ToDictionary(a => (a.Year, a.Month));

IEnumerable<DateOnly> DaysOf(int year) =>
    Enumerable.Range(1, DateTime.IsLeapYear(year) ? 366 : 365).Select(day => new DateOnly(year, 1, 1).AddDays(day - 1));

string[] Ramadan(int year, IReadOnlyDictionary<(int Year, int Month), Anchor> maps) =>
    DaysOf(year).Where(date => GetHijriDate(date, maps).Month == 9).Select(d => Iso(d.ToDateTime(TimeOnly.MinValue))).ToArray();

var ramadan2026 = Ramadan(2026, empty);
var ramadan2026Generated = Ramadan(2026, generated2026Maps);
var ramadan2026Shifted = Ramadan(2026, shifted2026Maps);
var ramadan2027 = Ramadan(2027, empty);
var ramadan2030 = Ramadan(2030, empty);

var resolvedShifted2026 = new Dictionary<string, string>();
foreach (var date in DaysOf(2026))
{
    var r = GetHijriDate(date, shifted2026Maps);
    resolvedShifted2026[Iso(date.ToDateTime(TimeOnly.MinValue))] = $"{r.Year}-{r.Month}-{r.Day}";
}

static string RamadanHijriYearTitle(int hijriYear, int year) => $"Ramadan {hijriYear} / {year}";

// (g) BuildHijriDate / GetHijriDate samples with valid, ignored and invalid anchors.
var displaySamples = new List<object>();
void Sample(string name, Anchor? anchor, int gy, int gm, int gd)
{
    var date = new DateOnly(gy, gm, gd);
    var display = BuildHijriDate(anchor, date);
    object? resolved;
    string? resolveError = null;
    try
    {
        var maps = anchor is null ? empty : new Dictionary<(int Year, int Month), Anchor> { [(anchor.Year, anchor.Month)] = anchor };
        var r = GetHijriDate(date, maps);
        resolved = new { day = r.Day, month = r.Month, year = r.Year };
    }
    catch (Exception ex)
    {
        resolved = null;
        resolveError = ex.GetType().Name;
    }
    displaySamples.Add(new { name, date = Iso(date.ToDateTime(TimeOnly.MinValue)), anchor, display, resolved, resolveError });
}
